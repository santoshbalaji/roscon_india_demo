cmake_minimum_required(VERSION 3.18)
project(roscon_collision)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(urdf REQUIRED)
find_package(fcl REQUIRED)
find_package(assimp REQUIRED)
find_package(yaml-cpp REQUIRED)
option(ROSCON_ENABLE_CUDA "Build the CUDA sphere collision backend" ON)
include(CheckLanguage)
if(ROSCON_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES "75-real;75-virtual")
  endif()
  check_language(CUDA)
endif()
if(ROSCON_ENABLE_CUDA AND CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(gpu_spheres src/gpu_spheres.cu src/gpu_sdf.cu)
  set_target_properties(gpu_spheres PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  # Keep PTX so newer GPUs can JIT this code even with an older CUDA toolkit.
  if(NOT CMAKE_CUDA_ARCHITECTURES)
    set_target_properties(gpu_spheres PROPERTIES CUDA_ARCHITECTURES "75-real;75-virtual")
  endif()
  target_compile_options(gpu_spheres PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  target_link_libraries(gpu_spheres CUDA::cudart)
else()
  message(STATUS "CUDA backend unavailable; FCL remains usable, GPU requests return an error")
  add_library(gpu_spheres src/gpu_spheres_unavailable.cpp src/gpu_sdf_unavailable.cpp)
endif()
target_include_directories(gpu_spheres PUBLIC include)
target_link_libraries(gpu_spheres sphere_model)
add_library(sphere_model src/sphere_model.cpp src/sphere_reference.cpp src/sdf_model.cpp src/sdf_reference.cpp)
target_include_directories(sphere_model PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(sphere_model fcl)
add_executable(collision_node src/collision_node.cpp src/geometry.cpp src/sdf_visualization.cpp)
target_include_directories(collision_node PRIVATE include)
ament_target_dependencies(collision_node rclcpp ament_index_cpp sensor_msgs std_msgs std_srvs visualization_msgs urdf)
target_link_libraries(collision_node fcl assimp::assimp yaml-cpp gpu_spheres sphere_model)
target_compile_options(collision_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS collision_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE)
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_spheres test/test_spheres.cpp)
  target_link_libraries(test_spheres sphere_model gpu_spheres)
  ament_add_gtest(test_sdf test/test_sdf.cpp)
  target_link_libraries(test_sdf sphere_model gpu_spheres)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(collision_integration test/test_collision.py TIMEOUT 90
    ENV COLLISION_EXECUTABLE=$<TARGET_FILE:collision_node>)
endif()
# Reusable synchronous CPU SDF checker for IK candidate validation.
add_library(scene_checker SHARED src/scene_checker.cpp src/geometry.cpp)
target_include_directories(scene_checker PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
set_target_properties(sphere_model PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(scene_checker PUBLIC sphere_model fcl assimp::assimp yaml-cpp)
ament_target_dependencies(scene_checker PUBLIC ament_index_cpp visualization_msgs urdf)
install(TARGETS scene_checker sphere_model EXPORT collision_targets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(collision_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(ament_index_cpp visualization_msgs urdf fcl assimp yaml-cpp)
ament_package()
