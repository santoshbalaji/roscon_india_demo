cmake_minimum_required(VERSION 3.16)
project(roscon_gearbox_demo)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(roscon_planner REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(validate_cell src/validate_cell.cpp)
ament_target_dependencies(validate_cell roscon_planner ament_index_cpp)
target_link_libraries(validate_cell yaml-cpp)
install(TARGETS validate_cell DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config docs DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_test(NAME gearbox_pipeline COMMAND validate_cell ${CMAKE_CURRENT_SOURCE_DIR}/config)
  set_tests_properties(gearbox_pipeline PROPERTIES TIMEOUT 300)
endif()
ament_package()
