cmake_minimum_required(VERSION 3.16)
project(roscon_planner)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(roscon_kinematics REQUIRED)
find_package(roscon_collision REQUIRED)
find_package(roscon_msgs REQUIRED)
find_package(ompl REQUIRED)
find_package(stomp REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(interactive_markers REQUIRED)
add_library(planner_core SHARED src/planner.cpp src/stomp_refinement.cpp src/chomp_refinement.cpp)
target_include_directories(planner_core PUBLIC include)
target_link_libraries(planner_core PUBLIC roscon_collision::scene_checker ompl::ompl stomp::stomp Eigen3::Eigen)
ament_target_dependencies(planner_core PUBLIC roscon_kinematics)
add_executable(planner_node src/node.cpp)
target_link_libraries(planner_node planner_core)
ament_target_dependencies(planner_node rclcpp roscon_msgs sensor_msgs geometry_msgs trajectory_msgs visualization_msgs std_msgs std_srvs interactive_markers)
install(TARGETS planner_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS planner_core LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(roscon_description REQUIRED)
  ament_add_gtest(test_planner test/test_planner.cpp)
  target_link_libraries(test_planner planner_core)
  set_tests_properties(test_planner PROPERTIES TIMEOUT 240)
  target_compile_definitions(test_planner PRIVATE
    ABB_URDF="${roscon_description_DIR}/../urdf/irb2600.urdf"
    SWEEP_URDF="${CMAKE_CURRENT_SOURCE_DIR}/test/one_joint.urdf"
    SWEEP_ENVIRONMENT="${CMAKE_CURRENT_SOURCE_DIR}/test/sweep_obstacle.yaml"
    TEST_ENVIRONMENT="${CMAKE_CURRENT_SOURCE_DIR}/config/environment.yaml")
  ament_add_pytest_test(planner_integration test/test_integration.py TIMEOUT 180
    ENV PLANNER_EXECUTABLE=$<TARGET_FILE:planner_node>
        PLANNER_URDF=${roscon_description_DIR}/../urdf/irb2600.urdf
        PLANNER_ENVIRONMENT=${CMAKE_CURRENT_SOURCE_DIR}/config/environment.yaml
        ROS_DOMAIN_ID=102)
endif()
ament_export_include_directories(include)
ament_export_libraries(planner_core)
ament_export_dependencies(roscon_kinematics roscon_collision Eigen3)
ament_package()
